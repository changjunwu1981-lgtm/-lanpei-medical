    let currentLang = 'zh';

    function changeLanguage(lang) {
        currentLang = lang;
        
        // 更新语言按钮状态
        document.querySelectorAll('.lang-btn').forEach(btn => {
            btn.classList.remove('bg-[#f97316]', 'text-white');
            if (btn.dataset.lang === lang) {
                btn.classList.add('bg-[#f97316]', 'text-white');
            }
        });
        
        // 更新页面标题
        updateTranslations();
    }

    function updateTranslations() {
        const t = translations[currentLang];
        
        // 更新所有翻译元素
        document.querySelectorAll('[data-i18n]').forEach(el => {
            const key = el.getAttribute('data-i18n');
            if (t[key]) {
                el.textContent = t[key];
            }
        });
        
        // 更新页脚
        const footerPlatform = document.getElementById('footer-platform');
        if (footerPlatform) footerPlatform.textContent = t.footer_platform;
        
        const footerDisclaimer = document.getElementById('footer-disclaimer');
        if (footerDisclaimer) footerDisclaimer.textContent = t.footer_disclaimer;
        
        // 更新CTA区域
        const ctaTitle = document.getElementById('cta-title');
        if (ctaTitle) ctaTitle.textContent = t.cta_title;
        
        const priceTitle = document.getElementById('price-title');
        if (priceTitle) priceTitle.innerHTML = '<i class="fas fa-tags mr-2 text-orange-500"></i>' + t.original_vs_generic;
        
        const priceRef = document.getElementById('price-reference');
        if (priceRef) priceRef.textContent = t.price_reference;
        
        // 更新价格表头
        const typeHeader = document.querySelector('.price-table th:nth-child(1)');
        if (typeHeader) typeHeader.textContent = t.type;
        const brandHeader = document.querySelector('.price-table th:nth-child(2)');
        if (brandHeader) brandHeader.textContent = t.brand;

        const priceHeader = document.querySelector('.price-table th:nth-child(4)');
        if (priceHeader) priceHeader.textContent = t.price;

奥希替尼（泰瑞沙）用法用量全解析｜第三代EGFR靶向药怎么用？

奥希替尼（商品名泰瑞沙/Tagrisso）是第三代EGFR酪氨酸激酶抑制剂，是非小细胞肺癌靶向治疗的里程碑药物。

【适应症】
①IB-IIIA期EGFR突变NSCLC术后辅助治疗（最长3年）
②III期不可切除NSCLC含铂放化疗后维持治疗
③EGFR敏感突变转移性NSCLC一线治疗
④联合培美曲塞+铂类一线治疗
⑤T790M突变阳性NSCLC二线治疗

【用法用量】
推荐剂量：80mg每日一次，口服
可随餐或空腹服用，每日固定时间
联合化疗时同样80mg每日一次
剂量调整：可减至40mg每日一次

【注意事项】
⚠️间质性肺病：出现呼吸困难立即停药
⚠️QT间期延长：定期监测心电图
⚠️心肌病：监测LVEF
⚠️用药前须基因检测确认EGFR突变
⚠️避免与强效CYP3A4诱导剂/抑制剂联用

【常见不良反应】
腹泻、皮疹、皮肤干燥、指甲毒性、恶心、疲劳

奥希替尼是全球首个获批的第三代EGFR-TKI，对脑转移疗效突出，可降低CNS进展或死亡风险83%。

卡博替尼（Cabometyx）用法用量全解析｜多靶点TKI肾癌肝癌靶向药怎么用？

卡博替尼（商品名Cabometyx）是多靶点酪氨酸激酶抑制剂，靶向MET、VEGFR2、AXL等多个信号通路，在肾癌、肝癌、甲状腺癌等多个癌种均有重要地位。

【适应症】
①晚期肾细胞癌：单药一线治疗或联合纳武利尤单抗
②晚期肝细胞癌：索拉非尼治疗后
③分化型甲状腺癌：RAI难治且系统治疗后进展
④神经内分泌瘤：既往治疗后进展

【用法用量】
单药：60mg每日一次，空腹服用
联合纳武利尤单抗：40mg每日一次
服药前1小时至服药后2小时禁食
整片吞服，不可压碎咀嚼

【剂量调整】
单药：60mg→40mg→20mg
联合：40mg→20mg→20mg隔日

【注意事项】
⚠️出血风险：严重出血需永久停药
⚠️胃肠道穿孔/瘘管：发生率约1%
⚠️高血压：需监测控制
⚠️手足综合征：注意皮肤护理
⚠️手术前至少停药3周
⚠️避免与CYP3A4强效诱导/抑制剂联用

【常见不良反应】
腹泻、疲劳、手足综合征、食欲减退、高血压、恶心

卡博替尼是NCCN 1类推荐药物，联合纳武利尤单抗已成为晚期肾癌一线标准治疗方案。

    function changeLanguage(lang) {
        currentLang = lang;
        
        // 更新语言按钮状态
        document.querySelectorAll('.lang-btn').forEach(btn => {
            btn.classList.remove('bg-[#f97316]', 'text-white');
            if (btn.dataset.lang === lang) {
                btn.classList.add('bg-[#f97316]', 'text-white');
            }
        });
        
        // 更新页面标题
        updateTranslations();
    }

维奈克拉（唯可来）用法用量全解析｜BCL-2抑制剂白血病怎么用？

维奈克拉（商品名唯可来/Venclexta）是全球首个BCL-2抑制剂，通过靶向BCL-2蛋白促进肿瘤细胞凋亡，为CLL和AML患者带来创新治疗选择。

【适应症】
①初治CLL/SLL：联合奥滨尤妥珠单抗
②经治CLL/SLL：联合利妥昔单抗
③伴17p缺失CLL单药：BTK抑制剂不适或失败
④新诊断AML（≥75岁或不适合强化化疗）：联合阿扎胞苷/地西他滨/低剂量阿糖胞苷

【用法用量】
CLL/SLL：5周剂量爬坡→400mg/日维持
  第1周20mg→第2周50mg→第3周100mg→第4周200mg→第5周400mg
AML：3天爬坡→400mg/日
  第1天100mg→第2天200mg→第3天起400mg
随餐口服，整片吞服

【注意事项】
