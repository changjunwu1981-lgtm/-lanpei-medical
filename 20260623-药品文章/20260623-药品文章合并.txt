    let currentLang = 'zh';

    function changeLanguage(lang) {
        currentLang = lang;
        
        // 更新语言按钮状态
        document.querySelectorAll('.lang-btn').forEach(btn => {
            btn.classList.remove('bg-[#f97316]', 'text-white');
            if (btn.dataset.lang === lang) {
                btn.classList.add('bg-[#f97316]', 'text-white');
            }
        });
        
        // 更新页面标题
        updateTranslations();
    }

    function updateTranslations() {
        const t = translations[currentLang];
        
        // 更新所有翻译元素
        document.querySelectorAll('[data-i18n]').forEach(el => {
            const key = el.getAttribute('data-i18n');
            if (t[key]) {
                el.textContent = t[key];
            }
        });
        
        // 更新页脚
        const footerPlatform = document.getElementById('footer-platform');
        if (footerPlatform) footerPlatform.textContent = t.footer_platform;
        
        const footerDisclaimer = document.getElementById('footer-disclaimer');
        if (footerDisclaimer) footerDisclaimer.textContent = t.footer_disclaimer;
        
        // 更新CTA区域
        const ctaTitle = document.getElementById('cta-title');
        if (ctaTitle) ctaTitle.textContent = t.cta_title;
        
        const priceTitle = document.getElementById('price-title');
        if (priceTitle) priceTitle.innerHTML = '<i class="fas fa-tags mr-2 text-orange-500"></i>' + t.original_vs_generic;
        
        const priceRef = document.getElementById('price-reference');
        if (priceRef) priceRef.textContent = t.price_reference;
        
        // 更新价格表头
        const typeHeader = document.querySelector('.price-table th:nth-child(1)');
        if (typeHeader) typeHeader.textContent = t.type;

    function changeLanguage(lang) {
        currentLang = lang;
        
        // 更新语言按钮状态
        document.querySelectorAll('.lang-btn').forEach(btn => {
            btn.classList.remove('bg-[#f97316]', 'text-white');
            if (btn.dataset.lang === lang) {
                btn.classList.add('bg-[#f97316]', 'text-white');
            }
        });
        
        // 更新页面标题
        updateTranslations();
    }

    let currentLang = 'zh';

    function changeLanguage(lang) {
        currentLang = lang;
        
        // 更新语言按钮状态
        document.querySelectorAll('.lang-btn').forEach(btn => {
            btn.classList.remove('bg-[#f97316]', 'text-white');
            if (btn.dataset.lang === lang) {
                btn.classList.add('bg-[#f97316]', 'text-white');
            }
        });
        
        // 更新页面标题
        updateTranslations();
    }

    function updateTranslations() {
        const t = translations[currentLang];
        
        // 更新所有翻译元素
        document.querySelectorAll('[data-i18n]').forEach(el => {
            const key = el.getAttribute('data-i18n');
            if (t[key]) {
                el.textContent = t[key];
            }
        });
        
        // 更新页脚
        const footerPlatform = document.getElementById('footer-platform');
        if (footerPlatform) footerPlatform.textContent = t.footer_platform;
        
        const footerDisclaimer = document.getElementById('footer-disclaimer');
        if (footerDisclaimer) footerDisclaimer.textContent = t.footer_disclaimer;
        
        // 更新CTA区域
        const ctaTitle = document.getElementById('cta-title');
        if (ctaTitle) ctaTitle.textContent = t.cta_title;
        
        const priceTitle = document.getElementById('price-title');
        if (priceTitle) priceTitle.innerHTML = '<i class="fas fa-tags mr-2 text-orange-500"></i>' + t.original_vs_generic;
        
        const priceRef = document.getElementById('price-reference');
        if (priceRef) priceRef.textContent = t.price_reference;
        
        // 更新价格表头

    function changeLanguage(lang) {
        currentLang = lang;
        
        // 更新语言按钮状态
        document.querySelectorAll('.lang-btn').forEach(btn => {
            btn.classList.remove('bg-[#f97316]', 'text-white');
            if (btn.dataset.lang === lang) {
                btn.classList.add('bg-[#f97316]', 'text-white');
            }
        });
        
        // 更新页面标题
        updateTranslations();
    }

地拉罗司（Deferasirox / Defrijet）——口服祛铁治疗的地中海贫血"铁过载克星"

【导语】
对于输血依赖型地中海贫血、骨髓增生异常综合征（MDS）等长期输血的患者来说，**铁过载**是悬在头顶的"达摩克利斯之剑"——铁沉积在心脏、肝脏、胰腺，可能导致心衰、肝硬化、糖尿病，甚至死亡。

今天给大家介绍全球第一个、也是应用最广的口服祛铁药物——**地拉罗司（Deferasirox）**。

---

## 一、药品简介

地拉罗司是一种**口服铁螯合剂**，通过与体内过量铁离子结合形成稳定的螯合物，经粪便排出体外，从而降低体内铁负荷。

该药由诺华（Novartis）原研，商品名为**Exjade（恩瑞格）**，2005年获FDA批准。印度Glenmark等多家厂家生产仿制版**Defrijet**。

**重磅消息**：中国国家药品监督管理局于2024年6月批准**地拉罗司颗粒（文帝平®）**在国内上市，是国内首个且唯一获批治疗铁过载的颗粒剂，特别适合儿童患者。

---

## 二、适应症

✅ **输血依赖型慢性铁过载**：2岁及以上患者（每月浓缩红细胞≥7mL/kg所致）
✅ **其他输血依赖性疾病**（MDS、再生障碍性贫血、镰状细胞病等）所致铁过载
✅ **非输血依赖性地中海贫血综合征（NTDT）**：10岁及以上患者

---

## 三、用法用量

**输血依赖型铁过载（TDT）**：
- 起始剂量：**14mg/kg/日**，口服，每日1次
- 每月接受<7mL/kg者：起始可考虑**7mg/kg/日**
- 每月接受>14mL/kg者：起始可考虑**21mg/kg/日**

**非输血依赖型地中海贫血综合征（NTDT）**：
- 起始剂量：**7mg/kg/日**
- 基线LIC>15mg Fe/g干重者，4周后可考虑**增至20mg/kg/日**
- **最大剂量不超过20mg/kg/日**

**服用方法**：
- ⚠️ **空腹服用**，餐前**至少30分钟**
- ⚠️ **不可咀嚼或整片吞服**，需先分散
- 将片剂投入**水、橙汁或苹果汁**中搅拌至形成均匀混悬液后立即饮服
  - 剂量<1g：约100mL液体
  - 剂量≥1g：约200mL液体
- 饮服后用少量液体冲洗容器一并饮下
- ⚠️ **不可与含铝抗酸药同时服用**

**剂量调整**：
- 治疗中每月监测血清铁蛋白
- 铁蛋白<300μg/L：**暂停治疗**，并检测LIC
- LIC<3mg Fe/g干重：**中断治疗**

---

## 四、重要注意事项

⚠️ **黑框警告：肾/肝功能衰竭、胃肠道出血**：
地拉罗司可能引起**急性肾损伤、肝毒性、胃肠道出血**等严重甚至致命的不良反应。

**治疗前检查**：
- 血清肌酐（重复2次）+ eGFR
- 肝功能（ALT/AST/胆红素）
- 血清铁蛋白
- 听力、眼科基线检查

**治疗中监测**：**每月**监测肾肝功能和血常规

⚠️ **常见不良反应**（>5%）：腹泻、呕吐、恶心、腹痛、皮疹、血清肌酐升高

⚠️ **严重皮肤反应**：Stevens-Johnson综合征（SJS）等——疑似即**永久停药**

⚠️ **绝对禁忌**：
- eGFR<40
- 一般状况差
- 高危MDS
- 晚期恶性肿瘤
- 血小板<50×10⁹/L
- 已知过敏

⚠️ **不可联用**：其他铁螯合剂（去铁胺、去铁酮）、含铝抗酸药
